SATHVIKA'S RESTAURANT — MYSQL WORKBENCH COMMANDS

1) Create database and select it
CREATE DATABASE IF NOT EXISTS food_ordering;
USE food_ordering;

2) Create menu table
CREATE TABLE IF NOT EXISTS food_items (
  food_id INT AUTO_INCREMENT PRIMARY KEY,
  food_name VARCHAR(120) NOT NULL,
  category VARCHAR(60) NOT NULL,
  price DECIMAL(10,2) NOT NULL,
  description TEXT,
  availability TINYINT(1) NOT NULL DEFAULT 1,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

3) Create order tables
CREATE TABLE IF NOT EXISTS orders (
  order_id INT AUTO_INCREMENT PRIMARY KEY,
  customer_name VARCHAR(120) NOT NULL,
  phone VARCHAR(20) NOT NULL,
  delivery_address TEXT NOT NULL,
  payment_method VARCHAR(40) NOT NULL,
  total_amount DECIMAL(10,2) NOT NULL,
  order_status VARCHAR(30) NOT NULL DEFAULT 'Placed',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE IF NOT EXISTS order_items (
  order_item_id INT AUTO_INCREMENT PRIMARY KEY,
  order_id INT NOT NULL,
  food_id INT NULL,
  food_name VARCHAR(120) NOT NULL,
  unit_price DECIMAL(10,2) NOT NULL,
  quantity INT NOT NULL,
  subtotal DECIMAL(10,2) NOT NULL,
  FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE,
  FOREIGN KEY (food_id) REFERENCES food_items(food_id) ON DELETE SET NULL
);

9) Delete a record (be careful; this is permanent)
DELETE FROM food_items WHERE food_id=6;

10) Useful filters and counts
SELECT * FROM food_items WHERE availability=1;
SELECT * FROM food_items WHERE category='Indian';
SELECT COUNT(*) AS total_food_items FROM food_items;
SELECT SUM(total_amount) AS total_order_value FROM orders;
